 oc create sa apple

# bind it to view role
oc policy add-role-to-user view -z apple

#### check role binding 
 oc describe rolebinding view
Name:         view
Labels:       <none>
Annotations:  <none>
Role:
  Kind:  ClusterRole
  Name:  view
Subjects:
  Kind            Name   Namespace
  ----            ----   ---------
  ServiceAccount  apple  devops
\########################################


#### Create pod using apple Service account 

apiVersion: v1
kind: Pod
metadata:
  labels:
    run: myapp
  name: myapp
spec:
  serviceAccountName: apple
  containers:
  - image: nginx
    name: myapp
    ports:
    - containerPort: 80

 oc create -f sa-pod.yaml

 oc describe pod myapp

#delete

 oc delete -f sa-pod.yaml
 oc delete sa apple
